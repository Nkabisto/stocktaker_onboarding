2. Combined automated and user retry mechanism
  /**
   * SERVER-SIDE: Express Controller
   * This handles "Optimistic Concurrency" automatically using retries.
   */
  async function bookSlot(req, res) {
    const { userId, occurrenceId } = req.body;
    const MAX_RETRIES = 3;

    for (let attempt = 1; attempt <= MAX_RETRIES; attempt++) {
      try {
        // attemptBooking should throw a specific 'VERSION_MISMATCH' error 
        // if the SQL 'UPDATE' affected 0 rows.
        await attemptBooking(userId, occurrenceId);
        
        return res.status(200).json({ 
          success: true, 
          message: 'Booking confirmed' 
        });

      } catch (error) {
        const isConflict = error.code === 'VERSION_MISMATCH';
        const canRetry = attempt < MAX_RETRIES;

        if (isConflict && canRetry) {
          // Log the collision and wait (Exponential Backoff: 100ms, 200ms...)
          console.warn(`Concurrency collision on attempt ${attempt}. Retrying...`);
          await new Promise(res => setTimeout(res, 100 * attempt));
          continue;
        }

        // If we are here, it's either a real error (DB down) or we ran out of retries
        const status = isConflict ? 409 : 500;
        const message = isConflict 
          ? 'The slot was taken by someone else. Please try again or pick another.' 
          : 'Internal server error';

        return res.status(status).json({ error: message, code: error.code });
      }
    }
  }

  /**
   * CLIENT-SIDE: React Component Function
   * This handles the "User-Level" retry logic.
   */
  const handleBooking = async () => {
    setLoading(true);
    setError(null);

    try {
      const response = await api.bookSlot(occurrenceId);
      setSuccess(true);
    } catch (error) {
      // 409 Conflict means even after server retries, the version changed or capacity was hit
      if (error.response?.status === 409) {
        setError({
          title: "Booking Conflict",
          message: "It looks like other people are booking this at the same time. Would you like to try one more time?",
          canRetry: true
        });
      } else {
        setError({
          title: "Error",
          message: "Something went wrong. Please try again later.",
          canRetry: false
        });
      }
    } finally {
      setLoading(false);
    }
  };


3. Slot generation algorithm 

  -- Query 1: Get slots for a specific date range
  SELECT * FROM event_occurrences
  WHERE start_time BETWEEN :start_date AND :end_date
    AND is_active = TRUE
    AND current_bookings < capacity_limit;

  -- Query 2: Get slots for a specific event type
  SELECT * FROM event_occurrences eo
  JOIN event_templates et on eo.template_id = et.id
  WHERE et.title = 'Interview'
    AND eo.start_time >= CURRENT_DATE
    AND eo.current_bookings < eo.capacity_limit;

  -- Query 3: Get user's bookings
  SELECT * FROM event_registrations
  WHERE user_id = :user_id
  ORDER BY registered_at DESC;
